{{/* Builds everything the home page renders from the two CollegeFootballData requests (rankings and games). No other endpoints are used. The head-to-head page reuses the same two requests, so a build fetches each only once. Returns a dict with: seasonYear - the season being shown pollWeek - the week of the AP poll in use (0 when there is none yet) pollIsFinal - true when the poll in use is the postseason (final) poll teams - the AP Top 25 in rank order, each with rank, school, conference and rankedGamesLeft teamBySchool - the same entries keyed by school games - remaining games between two ranked teams, by kickoff weekGroups - one entry per week of .games, in order, each with a label and the games in it headliner - the remaining game with the lowest combined AP rank (earliest kickoff wins ties), or false when none remain headlinerRankSum - that game's combined AP rank */}} {{ $cfbd := partial "cfbd-context.html" . }} {{ $seasonYear := $cfbd.seasonYear }} {{ $rankingWeeks := partial "cfbd-season-fetch.html" (dict "cfbd" $cfbd "endpoint" "rankings") }} {{ $allGames := partial "cfbd-season-fetch.html" (dict "cfbd" $cfbd "endpoint" "games") }} {{/* Use the most recent AP Top 25. */}} {{ $apPoll := partial "latest-poll.html" (dict "rankingWeeks" $rankingWeeks "pollPattern" "^AP Top 25$") }} {{ $apRanks := $apPoll.ranks }} {{ $pollWeek := $apPoll.week }} {{ $pollIsFinal := $apPoll.isFinal }} {{ $rankedSchools := slice }} {{ $rankBySchool := dict }} {{ range $apRanks }} {{ $rankedSchools = $rankedSchools | append .school }} {{ $rankBySchool = merge $rankBySchool (dict .school .rank) }} {{ end }} {{ $games := slice }} {{ range sort (where $allGames "completed" false) "startDate" }} {{ if and (in $rankedSchools .homeTeam) (in $rankedSchools .awayTeam) }} {{ $games = $games | append . }} {{ end }} {{ end }} {{ $teams := slice }} {{ $teamBySchool := dict }} {{ range $apRanks }} {{ $rankedGamesLeft := add (len (where $games "homeTeam" .school)) (len (where $games "awayTeam" .school)) }} {{ $team := dict "rank" .rank "school" .school "conference" .conference "rankedGamesLeft" $rankedGamesLeft }} {{ $teams = $teams | append $team }} {{ $teamBySchool = merge $teamBySchool (dict .school $team) }} {{ end }} {{ $weekGroups := slice }} {{ $previousWeekKey := "" }} {{ range $games }} {{ $weekKey := printf "%s-%d" .seasonType (int .week) }} {{ if ne $weekKey $previousWeekKey }} {{ $label := cond (eq .seasonType "postseason") "Postseason" (printf "Week %d" (int .week)) }} {{ $weekGames := where (where $games "seasonType" .seasonType) "week" .week }} {{ $weekGroups = $weekGroups | append (dict "label" $label "games" $weekGames) }} {{ $previousWeekKey = $weekKey }} {{ end }} {{ end }} {{ $headliner := false }} {{ $headlinerRankSum := 0 }} {{ range $games }} {{ $rankSum := add (index $rankBySchool .homeTeam) (index $rankBySchool .awayTeam) }} {{ if or (not $headliner) (lt $rankSum $headlinerRankSum) }} {{ $headliner = . }} {{ $headlinerRankSum = $rankSum }} {{ end }} {{ end }} {{ return dict "seasonYear" $seasonYear "pollWeek" $pollWeek "pollIsFinal" $pollIsFinal "teams" $teams "teamBySchool" $teamBySchool "games" $games "weekGroups" $weekGroups "headliner" $headliner "headlinerRankSum" $headlinerRankSum }}
